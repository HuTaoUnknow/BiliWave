cmake_minimum_required(VERSION 3.21)
project(BiliWave VERSION 1.11.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(BILI_ENABLE_LYRICS "Include lyric configuration and recognition features" ON)
option(BILI_ENABLE_RECOGNITION "Include local OCR, alignment and lyric import" ON)
option(BILI_ENABLE_ALIGNMENT "Include legacy audio alignment (not shipped in current editions)" OFF)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Multimedia Network Sql Test)
add_library(bili_core STATIC src/library.cpp src/lrc.cpp src/player.cpp src/autoducking.cpp src/chinese_lyrics.qrc)
target_include_directories(bili_core PUBLIC src)
target_link_libraries(bili_core PUBLIC Qt6::Widgets Qt6::Multimedia Qt6::Sql)
if(WIN32)
  target_link_libraries(bili_core PUBLIC ole32 uuid)
endif()
target_compile_definitions(bili_core PUBLIC BILI_ENABLE_LYRICS=$<BOOL:${BILI_ENABLE_LYRICS}>
  BILI_ENABLE_RECOGNITION=$<BOOL:${BILI_ENABLE_RECOGNITION}>
  BILI_ENABLE_ALIGNMENT=$<BOOL:${BILI_ENABLE_ALIGNMENT}> BILI_VERSION="${PROJECT_VERSION}")
set(BILI_UI src/window.cpp src/progressbar.cpp src/mainwindow.cpp src/playbackpage.cpp src/searchservice.cpp src/services.cpp src/onlinelyrics.cpp src/lyricworkflow.cpp src/streamservice.cpp src/theme.cpp src/desktopintegration.cpp src/ui.qrc)
add_executable(BiliWave WIN32 src/main.cpp ${BILI_UI})
if(WIN32)
  configure_file(src/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/biliwave.rc @ONLY)
  set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/biliwave.rc PROPERTIES
    OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/icons/biliwave.ico")
  target_sources(BiliWave PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/biliwave.rc)
endif()
target_link_libraries(BiliWave PRIVATE bili_core Qt6::Network)
if(MINGW)
  target_compile_options(bili_core PRIVATE -Wall -Wextra)
  target_compile_options(BiliWave PRIVATE -Wall -Wextra)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(workbench_tests tests/workbench_tests.cpp ${BILI_UI})
  target_link_libraries(workbench_tests PRIVATE bili_core Qt6::Network Qt6::Test)
  add_test(NAME workbench_tests COMMAND workbench_tests -platform windows -o workbench-results.txt,txt)
  set_tests_properties(workbench_tests PROPERTIES TIMEOUT 60)
  add_executable(online_lyrics_tests tests/online_lyrics_tests.cpp src/onlinelyrics.cpp)
  target_link_libraries(online_lyrics_tests PRIVATE bili_core Qt6::Network Qt6::Test)
  add_test(NAME online_lyrics_tests COMMAND online_lyrics_tests -platform windows -o online-lyrics-results.txt,txt)
  set_tests_properties(online_lyrics_tests PROPERTIES TIMEOUT 30)
  add_executable(ducking_tests tests/ducking_tests.cpp)
  target_link_libraries(ducking_tests PRIVATE bili_core Qt6::Test)
  add_test(NAME ducking_tests COMMAND ducking_tests -platform windows -o ducking-results.txt,txt)
  set_tests_properties(ducking_tests PROPERTIES TIMEOUT 30)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE bili_core Qt6::Test)
  add_test(NAME core_tests COMMAND core_tests -platform offscreen -o core-results.txt,txt)
  if(BILI_ENABLE_LYRICS AND BILI_ENABLE_RECOGNITION)
  add_executable(import_fixture tests/import_fixture.cpp)
  target_link_libraries(import_fixture PRIVATE Qt6::Core)
  add_executable(integration_tests tests/integration_tests.cpp ${BILI_UI})
  add_dependencies(integration_tests import_fixture)
  target_link_libraries(integration_tests PRIVATE bili_core Qt6::Network Qt6::Test)
  add_test(NAME integration_tests COMMAND integration_tests -platform windows -o integration-results.txt,txt)
  add_executable(redesign_tests tests/redesign_tests.cpp ${BILI_UI})
  target_link_libraries(redesign_tests PRIVATE bili_core Qt6::Network Qt6::Test)
  add_test(NAME redesign_tests COMMAND redesign_tests -platform windows -o redesign-results.txt,txt)
  set_tests_properties(core_tests integration_tests redesign_tests PROPERTIES TIMEOUT 90)
  add_executable(lyric_workflow_tests tests/lyric_workflow_tests.cpp src/lyricworkflow.cpp src/services.cpp src/onlinelyrics.cpp src/theme.cpp src/ui.qrc)
  target_link_libraries(lyric_workflow_tests PRIVATE bili_core Qt6::Network Qt6::Test)
  add_test(NAME lyric_workflow_tests COMMAND lyric_workflow_tests -platform windows -o lyric-workflow-results.txt,txt)
  set_tests_properties(lyric_workflow_tests PROPERTIES TIMEOUT 300)
  add_executable(live_lyrics tests/live_lyrics.cpp src/services.cpp src/onlinelyrics.cpp)
  target_link_libraries(live_lyrics PRIVATE bili_core Qt6::Network)
  add_executable(live_search tests/live_search.cpp src/searchservice.cpp)
  target_link_libraries(live_search PRIVATE bili_core Qt6::Network)
  add_executable(live_stream tests/live_stream.cpp src/streamservice.cpp src/services.cpp src/onlinelyrics.cpp)
  target_link_libraries(live_stream PRIVATE bili_core Qt6::Network)
  else()
    add_executable(lite_tests tests/lite_tests.cpp ${BILI_UI})
    target_link_libraries(lite_tests PRIVATE bili_core Qt6::Network Qt6::Test)
    add_test(NAME lite_tests COMMAND lite_tests -platform windows -o lite-results.txt,txt)
    set_tests_properties(lite_tests PROPERTIES TIMEOUT 90)
  endif()
endif()
